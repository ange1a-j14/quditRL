# Recipe (see EXPERIMENTS.md):
#   - keep the rollout shallow: seq_len ~ 2-3*d (deep rollouts collapse to 1/d^2)
#   - target-difficulty curriculum for d>=4 (escapes the 1/d^2 minimum)
#   - cosine LR decay + grad clipping are on by default in the trainer
# curriculum end depth defaults to seq_len.

# d=3 validated baseline (no curriculum needed): ~0.996 fidelity
python research/train.py --algo amortized --d 3 --target haar --hidden 512  --seq-len 6  --batch-targets 256 --amortized-iters 20000 --lr 1e-3 --loss infidelity --seed 0

# d=4..8: shallow seq_len=3*d + curriculum, hidden/iters scaled with dimension
python research/train.py --algo amortized --d 4 --target haar --hidden 1024 --seq-len 12 --batch-targets 256 --amortized-iters 30000 --lr 1e-3 --loss infidelity --target-curriculum --curriculum-frac 0.5 --curriculum-start-pulses 1 --seed 0
